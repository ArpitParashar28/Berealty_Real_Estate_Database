| Listing | CREATE TABLE `Listing` (
  `listing_id` int NOT NULL AUTO_INCREMENT,
  `property_id` int NOT NULL,
  `listing_type` enum('Sale','Rent') NOT NULL,
  `list_price` decimal(12,2) NOT NULL,
  `start_date` date NOT NULL,
  `end_date` date DEFAULT NULL,
  `status` enum('Active','Under Offer','Closed','Withdrawn') NOT NULL DEFAULT 'Active',

  UNIQUE KEY `uq_listing_one_open_per_property` (`open_property_id`),
  KEY `idx_listing_status_date` (`status`,`start_date`),
  KEY `idx_listing_property_type` (`property_id`,`listing_type`),
  CONSTRAINT `fk_listing_property` FOREIGN KEY (`property_id`) REFERENCES `Property` (`property_id`) ON DELETE RESTRICT ON UPDATE RESTRICT,
  CONSTRAINT `chk_listing_dates` CHECK (((`end_date` is null) or (`end_date` >= `start_date`))),
  CONSTRAINT `chk_listing_price` CHECK ((`list_price` > 0))
) ENGINE=InnoDB AUTO_INCREMENT=16 DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci |
